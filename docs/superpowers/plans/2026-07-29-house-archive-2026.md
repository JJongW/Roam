`ha-extract.py`를 스크래치패드에 만든다. **2패스**다.

```python
# HOUSE ARCHIVE 도면 PNG 2장 → 합쳐진 좌표계의 부스 사각형. 일회용, 커밋 금지.
#   python3 ha-extract.py > ha-rects.json
import json, sys
from PIL import Image

LEFT = "/Users/sinjong-won/ted.urssu/Roam/public/하우스아카이브_도면 좌측.png"
RIGHT = "/Users/sinjong-won/ted.urssu/Roam/public/하우스아카이브_도면 우측.png"
BG = (249, 254, 212)
DX, DY = 1314, 8          # 우측 → 좌측 좌표계. 겹침 0.

THEME = {                  # 도면 채우기색 → 테마
    (242, 176, 231): "collect",
    (248, 232, 115): "gather",
    (212, 200, 167): "make",
    (198, 241, 101): "rest",
    (241, 181, 156): "explore",
}


def near(c, t, tol=14):
    return all(abs(c[i] - t[i]) <= tol for i in range(3))


def is_fill(c):
    if near(c, BG):
        return False
    r, g, b = c[:3]
    return not (r > 235 and g > 235 and b > 235) and not (r < 90 and g < 90 and b < 90)


def quant(c):
    return (c[0] // 24, c[1] // 24, c[2] // 24)


def labelled(px, W, H, keep):
    """keep(color)->key 가 같은 인접 픽셀을 하나로 묶어 bbox를 낸다."""
    parent = {}

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    prev, runs = [], []
    for y in range(H):
        cur = []
        x = 0
        while x < W:
            k = keep(px[x, y])
            if k is not None:
                x0 = x
                while x + 1 < W and keep(px[x + 1, y]) == k:
                    x += 1
                rid = len(runs)
                runs.append((y, x0, x, px[x0, y]))
                parent[rid] = rid
                cur.append((x0, x, k, rid))
            x += 1
        for x0, x1, k, rid in cur:
            for p0, p1, pk, prid in prev:
                if pk == k and x0 <= p1 and p0 <= x1:
                    union(prid, rid)
        prev = cur

    agg = {}
    for rid, (y, x0, x1, c) in enumerate(runs):
        r = find(rid)
        a = agg.setdefault(r, [10**9, 10**9, -1, -1, 0, {}])
        a[0] = min(a[0], x0); a[1] = min(a[1], y)
        a[2] = max(a[2], x1); a[3] = max(a[3], y)
        a[4] += x1 - x0 + 1
        a[5][c] = a[5].get(c, 0) + (x1 - x0 + 1)
    return agg


def coloured(path):
    """패스 1 — 채우기색이 있는 부스(5테마 + CENTER)."""
    im = Image.open(path).convert("RGB"); W, H = im.size; px = im.load()
    agg = labelled(px, W, H, lambda c: quant(c) if is_fill(c) else None)
    out = []
    for a in agg.values():
        w, h = a[2] - a[0] + 1, a[3] - a[1] + 1
        if w < 14 or h < 14 or a[4] < 300 or a[4] / (w * h) < 0.45:
            continue
        col = max(a[5].items(), key=lambda kv: kv[1])[0]
        theme = next((t for c, t in THEME.items() if near(col, c, 18)), None)
        out.append({"x": a[0], "y": a[1], "w": w, "h": h,
                    "cat": theme, "rgb": list(col)})
    return out, (W, H)


def tables(path):
    """패스 2 — 테이블 마켓. 채우기가 배경색이고 검은 테두리에 갇힌 영역."""
    im = Image.open(path).convert("RGB"); W, H = im.size; px = im.load()
    agg = labelled(px, W, H, lambda c: 0 if near(c, BG) else None)
    out = []
    for a in agg.values():
        w, h = a[2] - a[0] + 1, a[3] - a[1] + 1
        # 바깥 거대 배경은 제외. 테이블은 ~24×48 또는 ~48×24.
        if not (400 < a[4] < 20000) or w < 18 or h < 18 or w > 90 or h > 90:
            continue
        out.append({"x": a[0], "y": a[1], "w": w, "h": h, "cat": "table"})
    return out


left, lsize = coloured(LEFT)
right, _ = coloured(RIGHT)
left += tables(LEFT)
right += tables(RIGHT)
for r in right:
    r["x"] += DX; r["y"] += DY

# 이음매(x ≈ 1314)에서 잘린 부스 병합: 좌측 조각이 이미지 오른쪽 끝에 닿고,
# 우측 조각이 이음매에 닿으며, y 구간이 60% 이상 겹치면 같은 부스다.
used = set()
for a in left:
    if a["x"] + a["w"] < lsize[0] - 2:
        continue
    for j, b in enumerate(right):
        if j in used or b["x"] > DX + 2:
            continue
        oy = min(a["y"] + a["h"], b["y"] + b["h"]) - max(a["y"], b["y"])
        if oy > min(a["h"], b["h"]) * 0.6:
            a["w"] = b["x"] + b["w"] - a["x"]
            a["h"] = max(a["h"], b["h"])
            a["cat"] = a["cat"] or b["cat"]
            a["seam"] = True
            used.add(j)
            break
rects = left + [b for j, b in enumerate(right) if j not in used]
rects.sort(key=lambda r: (r["y"], r["x"]))
json.dump({"leftSize": lsize, "offset": [DX, DY], "rects": rects},
          sys.stdout, ensure_ascii=False)
```

```js
import { readFileSync } from "node:fs";
import assert from "node:assert/strict";
const fp = JSON.parse(readFileSync(process.argv[2], "utf8"));
const b = fp.booths;

assert.equal(b.length, 104, "부스 수");
assert.equal(new Set(b.map((x) => x.code)).size, 104, "코드 유일");
assert.equal(fp.unitsPerMeter, 32, "스케일");

// CENTER 5개도 각자 테마 cat을 갖는다(kind만 facility). 검출 단계에서는 색이
// THEME에 없어 미분류로 나오지만, 최종 JSON에서는 테마가 붙는다.
//   collect 15+H01+CTR=17, gather 14+H02+CTR=16, make 13+H03+CTR=15,
//   rest 13+H04+CTR=15, explore 13+H05+CTR=15, table 26 → 합 104
const cats = {};
for (const x of b) cats[x.cat] = (cats[x.cat] ?? 0) + 1;
assert.deepEqual(cats, { collect: 17, gather: 16, make: 15, rest: 15, explore: 15, table: 26 }, "테마 분포");
assert.equal(Object.values(cats).reduce((a, c) => a + c, 0), 104, "분포 합");
assert.equal(b.filter((x) => x.kind === "facility").length, 5, "CENTER 5개");

for (const x of b) {
  assert.ok(x.name && x.nameEn, `${x.code}: 이름 누락`);
  assert.ok(x.w > 0 && x.h > 0, `${x.code}: 크기 0 이하`);
  assert.ok(x.x >= 0 && x.y >= 0 && x.x + x.w <= fp.width && x.y + x.h <= fp.height,
    `${x.code}: 캔버스 이탈`);
}

let ov = 0;
for (let i = 0; i < b.length; i++)
  for (let j = i + 1; j < b.length; j++) {
    const a = b[i], c = b[j];
    if (a.x < c.x + c.w && c.x < a.x + a.w && a.y < c.y + c.h && c.y < a.y + a.h) {
```

**Interfaces:**

- [ ] **Step 1: 컬럼을 확인한다**

`src/lib/supabase/repository.ts`에서 매핑을 읽어 컬럼명을 맞춘다: `mapExhibition`(~190행), `mapHall`(~200행), `mapCategory`(~210행), `mapBooth`(~227행), `BOOTH_LIST_COLS`(~225행). SIF의 `0026_sif_booth_fixes.sql`이 검증된 본이므로 그 INSERT 컬럼 목록을 그대로 따른다.

- [ ] **Step 2: 마이그레이션을 쓴다**

번호는 기존 최대(`0026_sif_booth_fixes.sql`) 다음인 `0027`. SIF와 같은 방침을 따른다:

- 파일 전체를 `begin;` / `commit;`으로 감싼다 — 실행 방법(psql `-f` vs SQL 에디터)과 무관하게 가드가 트립하면 전부 롤백되게.
- 선행 가드 `do $$ ... end $$;`로 **재실행 안전성**을 확인한다. 이 마이그레이션은 신규 삽입이므로 가드는 "전시가 이미 있으면 중단"이 아니라 **참조 무결성**을 본다: `organizer`가 필요하면 존재 여부, 그리고 이 전시 slug가 다른 id로 이미 있는지.
- 모든 INSERT는 `on conflict (id) do update set ...`로 멱등하게.
- id는 시드와 동일하게 고정한다(`exh_house_archive_2026`, `hall_ha`, `cat_ha_*`, `ha_*`) — 그래야 mock과 운영이 같은 키를 쓴다.
- 부스 104행은 `insert into booth (...) values (...), (...), ...` 한 문장으로. 값은 `seed-house-archive.ts`가 만드는 것과 일치해야 한다.
